## ✅ Correct Version (for RNN on Dino Names)

Here’s the corrected version:

```python
class DinoDataset(Dataset):
    def __init__(self, names_list, char_to_ix):
        self.names = names_list  # list of strings
        self.char_to_ix = char_to_ix

    def __len__(self):
        return len(self.names)

    def __getitem__(self, idx):
        name = self.names[idx].strip().lower()

        # Convert characters to indices
        x_chars = [None] + list(name)            # X starts with None
        y_chars = list(name) + ['\n']            # Y ends with newline

        x_indices = [self.char_to_ix[c] if c else None for c in x_chars]
        y_indices = [self.char_to_ix[c] for c in y_chars]

        # Replace None with a valid start token index (e.g., vocab size)
        start_token_ix = len(self.char_to_ix)    # Let's use this as index for None
        x_indices = [start_token_ix if i is None else i for i in x_indices]

        return torch.tensor(x_indices), torch.tensor(y_indices)
```
